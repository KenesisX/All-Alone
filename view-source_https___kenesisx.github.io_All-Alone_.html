function blobify(grid,GX,GY){
const thr=2, seen=new Uint8Array(GX*GY), blobs=[], st=[];
for(let i=0;i<grid.length;i++){ if(seen[i]||grid[i]<thr)continue;
let n=0,sx=0,sy=0,minx=GX,maxx=0,miny=GY,maxy=0; st.length=0; st.push(i); seen[i]=1;
while(st.length){ const j=st.pop(), jx=j%GX, jy=(j/GX)|0;
n++; sx+=jx+0.5; sy+=jy+0.5;
if(jx<minx)minx=jx; if(jx>maxx)maxx=jx; if(jy<miny)miny=jy; if(jy>maxy)maxy=jy;
if(jx>0&&!seen[j-1]&&grid[j-1]>=thr){seen[j-1]=1;st.push(j-1);}
if(jx<GX-1&&!seen[j+1]&&grid[j+1]>=thr){seen[j+1]=1;st.push(j+1);}
if(jy>0&&!seen[j-GX]&&grid[j-GX]>=thr){seen[j-GX]=1;st.push(j-GX);}
if(jy<GY-1&&!seen[j+GX]&&grid[j+GX]>=thr){seen[j+GX]=1;st.push(j+GX);} }
if(n>=2)blobs.push({n,cx:sx/n/GX,cy:sy/n/GY,w:(maxx-minx+1)/GX,h:(maxy-miny+1)/GY}); }
return blobs;
}

/* pixel-level camera filters */
function applyFilter(g,w,h,f){
const W=g.canvas.width,H=g.canvas.height;
let img;try{img=g.getImageData(0,0,W,H);}catch(e){return;}
const d=img.data;
for(let i=0;i<d.length;i+=4){
const r=d[i],gr=d[i+1],b=d[i+2];
const lum=(r*.299+gr*.587+b*.114);
switch(f){
case 'thermal':{const t=lum/255;
let R,G,B;
if(t<.25){R=0;G=0;B=t*4*200;}
else if(t<.5){R=0;G=(t-.25)*4*255;B=200-(t-.25)*4*200;}
else if(t<.75){R=(t-.5)*4*255;G=255;B=0;}
else{R=255;G=255-(t-.75)*4*200;B=(t-.75)*4*255;}
d[i]=R;d[i+1]=G;d[i+2]=B;break;}
case 'night':{const v=clamp(lum*1.9,0,255);d[i]=v*.25;d[i+1]=v;d[i+2]=v*.3;break;}
case 'neg':d[i]=255-r;d[i+1]=255-gr;d[i+2]=255-b;break;
case 'red':{const v=clamp(lum*1.5,0,255);d[i]=v;d[i+1]=v*.12;d[i+2]=v*.12;break;}
case 'uv':{d[i]=clamp(b*1.3+lum*.3,0,255);d[i+1]=clamp(lum*.2,0,255);d[i+2]=clamp(r*1.1+120,0,255);break;}
case 'mono':{const v=clamp((255-lum)*1.1,0,255);d[i]=d[i+1]=d[i+2]=v;break;}
case 'pred':{const t=lum/255;d[i]=clamp(Math.sin(t*6.28)*127+128,0,255);
d[i+1]=clamp(Math.sin(t*6.28+2)*127+128,0,255);d[i+2]=clamp(Math.sin(t*6.28+4)*127+128,0,255);break;}
case 'xray':{const v=255-clamp(lum*1.4,0,255);d[i]=v*.7;d[i+1]=v*.85;d[i+2]=v;break;}
case 'aurora':{const t=lum/255;d[i]=clamp(t*80,0,255);d[i+1]=clamp(t*255,0,255);
d[i+2]=clamp(120+t*100,0,255);break;}
case 'multispec':{const t=lum/255; // visible-light false-colour band split (not real EM/IR sensing)
d[i]=clamp(Math.sin(t*3.14159)*255,0,255);
d[i+1]=clamp(Math.sin(t*3.14159+2.1)*255,0,255);
d[i+2]=clamp(Math.sin(t*3.14159+4.2)*255,0,255);break;}
}
}
if(f==='edge'){edgeSobel(d,W,H);}
g.putImageData(img,0,0);
}
